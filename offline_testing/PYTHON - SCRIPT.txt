def apply_layout(data):
    # Carica il dizionario dal JSON
    layout = json.loads(data)
    
    for item in layout["placements"]:
        obj_name = item["name"]
        
        # Verifica se l'oggetto esiste nella scena attuale
        if obj_name in bpy.data.objects:
            obj = bpy.data.objects[obj_name]
            
            # Applica le coordinate X e Y (mantenendo invariata l'altezza Z originaria)
            obj.location.x = item["x"]
            obj.location.y = item["y"]
            
            # Converte la rotazione da gradi in radianti e la applica sull'asse Z
            rad = math.radians(item["rotation_deg"])
            obj.rotation_euler.z = rad
            
            print(f"Applicato: {obj_name} -> Pos: ({item['x']}, {item['y']}), Rot Z: {item['rotation_deg']}°")
        else:
            print(f"Attenzione: L'oggetto '{obj_name}' non è stato trovato nella scena di Blender.")

# Esegui la funzione
apply_layout(json_data)
